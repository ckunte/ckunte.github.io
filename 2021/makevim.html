Make Vim

26 Dec 2021

Compiling Vim from source with python3 interpreter enabled:

#!/usr/bin/env bash
if ! [ -d "./vim" ] ; then
    git clone --depth=1 https://github.com/vim/vim.git
fi
if [ -d "./vim" ] ; then
    cd './vim/src' || return
    git pull --ff-only
    make distclean
    ./configure --enable-python3interp
    make
    echo "Run sudo make install in './vim/src' for a good compile."
fi
