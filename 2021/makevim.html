Make Vim

26 Dec 2021

Compiling Vim from source with python3 interpreter enabled:

#!/usr/bin/env bash
set -euo pipefail

VIM_DIR="./vim"
PREFIX="/usr/local"

if [[ ! -d "$VIM_DIR/.git" ]]; then
    git clone --depth=1 https://github.com/vim/vim.git "$VIM_DIR"
else
    git -C "$VIM_DIR" pull --ff-only
fi

cd "$VIM_DIR/src"

make distclean
./configure \
    --prefix="$PREFIX" \
    --enable-python3interp

make -j"$(nproc)"
sudo make install

echo "Vim installed successfully."
